<script>

// ----------------------------------------------------
// CREATE FLOATING METERS
// ----------------------------------------------------
function createFloatingPanel(id, title, defaultBottom = 20, defaultRight = 20) {
    const panel = document.createElement("div");
    panel.id = id;
    panel.style.cssText = `
        position:fixed;
        bottom:${defaultBottom}px;
        right:${defaultRight}px;
        width:240px;
        background:#0f0f0f;
        border:1px solid #333;
        border-radius:8px;
        padding:10px;
        font-size:13px;
        color:#ccc;
        z-index:99999;
        box-shadow:0 0 10px #000;
        cursor:move;
    `;

    panel.innerHTML = `
        <div style="font-weight:bold;color:#66ccff;margin-bottom:6px;">
            ${title}
            <span style="float:right;cursor:pointer;color:#888;" onclick="togglePanel('${id}')">−</span>
        </div>
        <div id="${id}-content"></div>
    `;

    makeDraggable(panel);
    document.body.appendChild(panel);
}

function togglePanel(id) {
    const content = document.getElementById(id + "-content");
    content.style.display = content.style.display === "none" ? "block" : "none";
}

// ----------------------------------------------------
// MEMORY GRAPH PANEL + GRAPH LOGIC
// ----------------------------------------------------
let memData = [];
const memMaxPoints = 60;

function createMemoryPanel() {
    const panel = document.createElement("div");
    panel.id = "memMeter";
    panel.style.cssText = `
        position:fixed;
        bottom:20px;
        right:800px;
        width:300px;
        height:200px;
        background:#0f0f0f;
        border:1px solid #333;
        border-radius:8px;
        padding:10px;
        font-size:13px;
        color:#ccc;
        z-index:99999;
        box-shadow:0 0 10px #000;
        cursor:move;
    `;

    panel.innerHTML = `
        <div style="font-weight:bold;color:#66ccff;margin-bottom:6px;">
            Memory Usage
            <span style="float:right;cursor:pointer;color:#888;" onclick="togglePanel('memMeter')">−</span>
        </div>
        <canvas id="memCanvas" width="280" height="160"></canvas>
    `;

    makeDraggable(panel);
    document.body.appendChild(panel);
}

function updateMemoryGraph() {
    if (!performance || !performance.memory) return;

    const used = performance.memory.usedJSHeapSize / (1024 * 1024);
    const total = performance.memory.totalJSHeapSize / (1024 * 1024);

    memData.push(used);
    if (memData.length > memMaxPoints) memData.shift();

    const canvas = document.getElementById("memCanvas");
    if (!canvas) return;
    const ctx = canvas.getContext("2d");

    const w = canvas.width;
    const h = canvas.height;

    ctx.clearRect(0, 0, w, h);

    if (memData.length < 2) return;

    const maxVal = Math.max(...memData) * 1.1;
    const minVal = Math.min(...memData) * 0.9;

    ctx.beginPath();
    memData.forEach((val, i) => {
        const x = (i / (memMaxPoints - 1)) * (w - 10) + 5;
        const norm = (val - minVal) / (maxVal - minVal || 1);
        const y = h - norm * (h - 10) - 5;

        if (i === 0) ctx.moveTo(x, y);
        else ctx.lineTo(x, y);
    });

    const gradient = ctx.createLinearGradient(0, 0, 0, h);
    gradient.addColorStop(0, "rgba(102,204,255,0.5)");
    gradient.addColorStop(1, "rgba(102,204,255,0)");

    ctx.lineWidth = 2;
    ctx.strokeStyle = "#66ccff";
    ctx.stroke();

    ctx.lineTo(w - 5, h - 5);
    ctx.lineTo(5, h - 5);
    ctx.closePath();
    ctx.fillStyle = gradient;
    ctx.fill();

    ctx.fillStyle = "#cccccc";
    ctx.font = "11px Arial";
    ctx.fillText(`Used: ${used.toFixed(1)} MB`, 10, 16);
    ctx.fillText(`Total: ${total.toFixed(1)} MB`, 10, 30);
}

setInterval(updateMemoryGraph, 2000);

// ----------------------------------------------------
// DRAGGABLE PANELS
// ----------------------------------------------------
function makeDraggable(el) {
    let offsetX = 0, offsetY = 0, dragging = false;

    el.addEventListener("mousedown", e => {
        dragging = true;
        offsetX = e.clientX - el.offsetLeft;
        offsetY = e.clientY - el.offsetTop;
    });

    document.addEventListener("mouseup", () => dragging = false);

    document.addEventListener("mousemove", e => {
        if (dragging) {
            el.style.left = (e.clientX - offsetX) + "px";
            el.style.top = (e.clientY - offsetY) + "px";
            el.style.bottom = "auto";
            el.style.right = "auto";
        }
    });
}

// ----------------------------------------------------
function createAllPanels() {
    createFloatingPanel("chunkMonitor", "Chunk Monitor", 20, 20);
    createFloatingPanel("perfMeter", "Performance Meter", 20, 280);
    createFloatingPanel("netMeter", "Network Meter", 20, 540);
    createMemoryPanel();
}

createAllPanels();

// Load pages.json and populate the right panel
async function loadPageList() {
    try {
        const res = await fetch("/amin/pages.json?cache=" + Date.now());
        const pages = await res.json();

        const container = document.getElementById("pageList");
        if (!container) return;

        container.innerHTML = "";

        pages.forEach(page => {
            const div = document.createElement("div");
            div.className = "pageItem";
            div.style.cssText = `
                padding:8px;
                margin-bottom:6px;
                background:#111;
                border:1px solid #333;
                border-radius:6px;
                cursor:pointer;
            `;
            div.textContent = page;

            div.onclick = () => loadPageStats(page);

            container.appendChild(div);
        });

    } catch (err) {
        console.error("Failed to load pages.json", err);
    }
}

// Load stats for a specific page
async function loadPageStats(page) {
    const statsBox = document.getElementById("statsBox");
    if (!statsBox) return;

    statsBox.innerHTML = `<div style="color:#66ccff;">Loading stats for ${page}...</div>`;

    try {
        const res = await fetch(`/api/${page.replace(".html", "")}-feedback?cache=` + Date.now());
        const data = await res.json();

        statsBox.innerHTML = `
            <div style="font-size:16px;color:#66ccff;margin-bottom:10px;">${page}</div>
            <div style="margin-bottom:6px;">Views: <span style="color:#fff;">${data.views || 0}</span></div>
            <div style="margin-bottom:6px;">Likes: <span style="color:#0f0;">${data.likes || 0}</span></div>
            <div style="margin-bottom:6px;">Dislikes: <span style="color:#f33;">${data.dislikes || 0}</span></div>
            <div style="margin-bottom:6px;">Feedback Count: <span style="color:#ff0;">${data.feedbackCount || 0}</span></div>
        `;
    } catch (err) {
        statsBox.innerHTML = `<div style="color:#f33;">Failed to load stats.</div>`;
    }
}

// Load global counters (index, cheat, total)
async function loadGlobalCounters() {
    try {
        const res = await fetch("/api/global-stats?cache=" + Date.now());
        const data = await res.json();

        document.getElementById("indexViews").textContent = data.index || 0;
        document.getElementById("cheatViews").textContent = data.cheat || 0;
        document.getElementById("totalViews").textContent = data.total || 0;

    } catch (err) {
        console.error("Failed to load global counters", err);
    }
}

// Load Top 10 pages
async function loadTop10() {
    try {
        const res = await fetch("/api/top10?cache=" + Date.now());
        const data = await res.json();

        const box = document.getElementById("top10Box");
        if (!box) return;

        box.innerHTML = "";

        data.forEach((item, i) => {
            const div = document.createElement("div");
            div.style.cssText = `
                padding:6px;
                margin-bottom:4px;
                background:#111;
                border:1px solid #333;
                border-radius:6px;
            `;
            div.innerHTML = `<span style="color:#66ccff;">${i + 1}.</span> ${item.page} — <span style="color:#0f0;">${item.views}</span> views`;

            box.appendChild(div);
        });

    } catch (err) {
        console.error("Failed to load Top 10", err);
    }
}

// Initialize admin panel
function initAdmin() {
    loadPageList();
    loadGlobalCounters();
    loadTop10();
}

// Auto-refresh every 10 seconds
setInterval(() => {
    loadGlobalCounters();
    loadTop10();
}, 10000);

// Run admin engine after DOM loads
document.addEventListener("DOMContentLoaded", initAdmin);

</script>
